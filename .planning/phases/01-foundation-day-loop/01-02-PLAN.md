simulation/defs/ (Resource scripts):
```gdscript
class_name BuildingTierDef extends Resource
@export var cost: int              # gold, > 0
@export var dawn_income: int       # 0 for non-economy buildings
@export var attack_range: float    # 0.0 for non-defense buildings (tower stats are data even with no enemies yet)
@export var attack_damage: int
func effect_line() -> String       # "+N gold at dawn" if dawn_income > 0; "Range R · Damage D" if attack_damage > 0; else ""

class_name BuildingDef extends Resource
@export var id: StringName          # &"house", &"tower"
@export var display_name: String    # "House"
@export var tiers: Array[BuildingTierDef]   # index 0 = tier I (the build), 1 = tier II ...
func max_tier() -> int
func tier_def(tier: int) -> BuildingTierDef   # 1-based; null when out of range
func tier_label(tier: int) -> String          # "House II" (Roman numerals I..V)

class_name BuildSpotDef extends Resource
@export var id: StringName          # &"house_1"
@export var position: Vector3       # world XZ; y = 0
@export var building_id: StringName # the ONE building type this spot allows (D-02)

class_name MapConfig extends Resource
@export var id: StringName
@export var display_name: String
@export var is_sandbox: bool = true           # D-04: sandbox/test maps never appear in the campaign
@export var starting_gold: int
@export var king_spawn: Vector3
@export var castle_position: Vector3
@export var buildings: Array[BuildingDef]     # building set this map uses
@export var spots: Array[BuildSpotDef]        # order is significant (tie-breaks, payout order)

class_name LoopTuning extends Resource
@export var coin_drip_interval: float = 0.2   # seconds per coin (D-05)
@export var interaction_radius: float = 2.5   # metres, XZ distance, inclusive

class_name KingDef extends Resource
@export var walk_speed: float = 5.0
@export var sprint_multiplier: float = 1.6
```

simulation/ (RefCounted, zero scene-tree/rendering dependency):
```gdscript
class_name SimEvents extends RefCounted
signal gold_changed(new_amount: int, delta: int)
signal building_built(spot_id: StringName, building_id: StringName, new_tier: int)  # new_tier 1 = constructed, >1 = upgraded
signal command_rejected(command: StringName, spot_id: StringName, reason: StringName)
signal phase_changed(old_phase: int, new_phase: int)

class_name Economy extends RefCounted
func _init(events: SimEvents, starting_gold: int) -> void
func get_gold() -> int
func can_afford(cost: int) -> bool      # cost >= 0 and gold >= cost
func try_spend(cost: int) -> bool       # false (no change) if cost < 0 or unaffordable
func grant(amount: int) -> bool         # false (no change) if amount < 0

class_name BuildingInstance extends RefCounted
var spot_id: StringName
var building_id: StringName
var tier: int

class_name BuildingSystem extends RefCounted
func _init(map: MapConfig, events: SimEvents) -> void
func spot_ids() -> Array[StringName]                 # MapConfig order
func get_spot(spot_id: StringName) -> BuildSpotDef   # null if unknown
func get_building_def_for_spot(spot_id: StringName) -> BuildingDef
func get_instance(spot_id: StringName) -> BuildingInstance   # null if empty
func current_tier(spot_id: StringName) -> int        # 0 if empty
func next_tier_def(spot_id: StringName) -> BuildingTierDef   # null at max tier or unknown spot
func next_action_cost(spot_id: StringName) -> int    # -1 when nothing can be bought
func nearest_spot_in_range(pos: Vector3, radius: float) -> StringName  # XZ distance <= radius; ties -> earlier MapConfig order; &"" if none
func apply_next_tier(spot_id: StringName) -> BuildingInstance  # NO validation; only CommandProcessor calls it

class_name RunManager extends RefCounted
enum RunPhase { DAY, NIGHT_TRANSITION, NIGHT, DAWN }
func _init(events: SimEvents) -> void
func get_phase() -> RunPhase
func is_build_allowed() -> bool          # phase == DAY
func tick(delta: float) -> void          # the only time source for the simulation

class_name BuildIntent extends RefCounted
var spot_id: StringName
func _init(target_spot: StringName) -> void

class_name CommandProcessor extends RefCounted
const OK := &"ok"; const NOT_DAY := &"not_day"; const UNKNOWN_SPOT := &"unknown_spot"
const MAX_TIER := &"max_tier"; const CANNOT_AFFORD := &"cannot_afford"
func _init(economy: Economy, buildings: BuildingSystem, run_manager: RunManager, events: SimEvents) -> void
func validate_build(spot_id: StringName) -> StringName   # read-only; check order: NOT_DAY, UNKNOWN_SPOT, MAX_TIER, CANNOT_AFFORD
func submit(intent: RefCounted) -> StringName             # dispatches on intent type; BuildIntent -> validate, try_spend, apply_next_tier

class_name RunContext extends RefCounted
var map: MapConfig; var tuning: LoopTuning; var events: SimEvents; var economy: Economy
var buildings: BuildingSystem; var run_manager: RunManager; var commands: CommandProcessor
func _init(map_config: MapConfig, loop_tuning: LoopTuning) -> void
```
